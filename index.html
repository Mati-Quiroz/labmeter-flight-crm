1. Esquema Relacional SQL (Simuladores Oficiales + Salas de Clases)

-- TABLA 1: PILOTOS Y CLIENTES (CRM)
CREATE TABLE pilots (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  full_name VARCHAR(150) NOT NULL,
  rut_passport VARCHAR(40) UNIQUE NOT NULL,
  email VARCHAR(150),
  phone VARCHAR(40),
  birth_date DATE NOT NULL, -- Requerido para trigger automático de Giftcard de cumpleaños
  license_type VARCHAR(80), -- Opcional (PPA, PCA, IFR, Aficionado)
  logged_flight_hours NUMERIC(8,1) DEFAULT 0,
  client_segment VARCHAR(50) NOT NULL, -- 'Piloto Privado','Comercial','Instructor','Aficionado','Visita Pedagógica','Visita Comercial'
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- TABLA 2: RECURSOS RESERVABLES (4 SIMULADORES + 4 SALAS DE CLASES/REUNIONES)
CREATE TABLE facilities_and_simulators (
  id VARCHAR(30) PRIMARY KEY,
  resource_type VARCHAR(20) NOT NULL, -- 'SIMULADOR' o 'SALA'
  name VARCHAR(120) NOT NULL,
  registration_code VARCHAR(40),      -- 'LAB-CH-134', 'PC-LAB-70', 'PC-LAB-64', 'NO CERTIFICADO' o Capacidad de Sala
  capacity_people INTEGER DEFAULT 2,  -- Ej: 15 (SALA 2), 6 (SALA 3), 9 (SALA LM Puente Alto)
  is_certified BOOLEAN DEFAULT TRUE,
  is_active BOOLEAN DEFAULT TRUE
);

-- TABLA 3: OPERADORES / INSTRUCTORES DE SIMULADOR
CREATE TABLE operators (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  full_name VARCHAR(120) NOT NULL,
  email VARCHAR(150) UNIQUE NOT NULL,
  base_weekly_days INTEGER DEFAULT 5,
  totp_secret VARCHAR(64) NOT NULL -- Semilla para validación de QR dinámico anti-fraude
);

-- TABLA 4: MATRIZ DE TURNOS Y CONTROL DE ASISTENCIA QR + UBER
CREATE TABLE shifts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  resource_id VARCHAR(30) REFERENCES facilities_and_simulators(id),
  operator_id UUID REFERENCES operators(id),
  shift_date DATE NOT NULL,
  start_time TIME NOT NULL, -- Jornada base 08:00 a 18:00
  end_time TIME NOT NULL,
  regular_hours NUMERIC(4,1) DEFAULT 0,
  extension_hours NUMERIC(4,1) DEFAULT 0,  -- Horas posteriores a las 18:00 hrs
  activation_hours NUMERIC(4,1) DEFAULT 0, -- Fines de semana o fuera de horario normal (tarifa mayor)
  requires_uber BOOLEAN DEFAULT FALSE,     -- TRUE cuando end_time está entre 21:00 y 06:00 hrs
  qr_checkin_token VARCHAR(64),
  qr_checkout_token VARCHAR(64),
  UNIQUE (resource_id, shift_date, start_time)
);

-- TABLA 5: RESERVAS DE BLOQUES DE SIMULADOR O SALAS
CREATE TABLE bookings (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  pilot_id UUID REFERENCES pilots(id),
  resource_id VARCHAR(30) REFERENCES facilities_and_simulators(id),
  shift_id UUID REFERENCES shifts(id),
  booking_date DATE NOT NULL,
  time_slot VARCHAR(40) NOT NULL,
  status VARCHAR(25) DEFAULT 'Pendiente',
  admin_notified BOOLEAN DEFAULT FALSE, -- Notificación a mquiroz@labmeter.cl; arios@labmeter.cl; vrios@labmeter.cl
  created_at TIMESTAMPTZ DEFAULT NOW()
);
